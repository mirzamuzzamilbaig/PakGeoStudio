var getFireConf = function(area_of_interest,fireDict,endTime) {

 // Satellite data
  var goes_16_data = goes16fdcf
    .filterDate(fireDict.start,endTime).select('Mask');

  var goes_17_data = goes17fdcf
    .filterDate(fireDict.start,endTime).select('Mask');

  // Conversion from mask codes to confidence values
  var confidence_codes_to_values = function(confidence) {
    return confidence.expression(
    '(conf==1.0) + (conf==2.0)*0.9 + (conf==3.0)*0.8 + (conf==4.0)*0.5 + (conf==5.0)*0.3 + (conf==6.0)*0.1',
      {conf: confidence});
  };

  var setPixCount = function(image) {
    var pixels = image.lte(35).rename('pixels')
      .reduceRegion({
        geometry: area_of_interest,
        reducer: ee.Reducer.sum().unweighted()
      }).getNumber('pixels');

    if (fireDict.minPix !== undefined) {
      var hrDiff = ee.Date(image.get('system:time_start'))
        .difference(fireDict.start,'hour').round();
      image = image.set('hrOffset',hrDiff);
    }

    return image.set('pixels',pixels);
  };

  var getGOESconf = function(goes_data) {
    var goes_confidence = goes_data
      .map(setPixCount).filter(ee.Filter.gt('pixels',0));
    var nImg = goes_confidence.size();

    if (fireDict.minPix !== undefined) {
      goes_confidence = goes_confidence.filter(ee.Filter.lt('hrOffset',24))
        .merge(goes_confidence.filter(ee.Filter.gte('hrOffset',24))
          .filter(ee.Filter.gt('pixels',fireDict.minPix)));
    }

    return ee.Image(ee.Algorithms.If(nImg.gt(0),
      goes_confidence
        .map(function(image) {
          return image.updateMask(image.lte(35))
          .mod(10).add(1).toInt();
        }).min().unmask(0),
      ee.Image(0))).rename('confidence');
  };

  var goes_16_max_confidence = confidence_codes_to_values(getGOESconf(goes_16_data));
  var goes_17_max_confidence = confidence_codes_to_values(getGOESconf(goes_17_data));

  var combined_confidence = ee.ImageCollection([
    goes_16_max_confidence,
    goes_17_max_confidence,
  ]).mean();

  var kernel = ee.Kernel.square(2000, 'meters', true);
  var smoothed_confidence = combined_confidence.clip(area_of_interest)
    .reduceNeighborhood({
      'reducer': ee.Reducer.mean(),
      'kernel': kernel,
      'optimization': 'boxcar'
    });

  return smoothed_confidence;
};

var getFirePerim = function(fireDict,area_of_interest,smoothed_confidence,confidence_cutoff) {
  var high_confidence = smoothed_confidence.gt(confidence_cutoff);

  var affected_areas = high_confidence
    .reduceToVectors({
      scale: 100,
      maxPixels: 1e10,
      geometry: area_of_interest,
      tileScale: fireDict.tileScale,
      bestEffort: true
    }).filter(ee.Filter.eq('label', 1));

  return affected_areas;
};

var getFirePerimImg = function(affected_areas, width) {
  var smooth = function(feature) {
    var max_error_meters = 200;
    return ee.Feature(feature).simplify(max_error_meters);
  };

  var affected_areas_smoothed = ee.FeatureCollection(affected_areas)
    .map(smooth);
  var affected_areas_smoothed_outline = ee.Image().byte()
    .paint({featureCollection: affected_areas_smoothed, width: width});

  return affected_areas_smoothed_outline;
};

var getDynamic = function(fireDict,area_of_interest) {

  var dnbr_dates = {
    'preS': ee.Date(fireDict.start).advance(-16,'day'),
    'postE': ee.Date(fireDict.end).advance(15,'day')
  };

  var dw = dynamic
    .filterDate(dnbr_dates.preS,dnbr_dates.postE)
    .filterBounds(area_of_interest);

  var classification = dw.select('label');
  var dwComposite = classification.reduce(ee.Reducer.mode());

  var probabilityBands = [
    'water', 'trees', 'grass', 'flooded_vegetation', 'crops', 'shrub_and_scrub',
    'built', 'bare', 'snow_and_ice'
  ];

  // Select probability bands.
  var probabilityCol = dw.select(probabilityBands);

  // Create an image with the average pixel-wise probability
  // of each class across the time-period.
  var meanProbability = probabilityCol.reduce(ee.Reducer.mean());

  var projection = ee.Projection('EPSG:3857').atScale(10);
  meanProbability = meanProbability.setDefaultProjection(projection);

  // Create the Top-1 Probability Hillshade.
  var top1Probability = meanProbability.reduce(ee.Reducer.max());
  var top1Confidence = top1Probability.multiply(100).int();
  var hillshade = ee.Terrain.hillshade(top1Confidence).divide(255);
  var rgbImage = dwComposite.visualize(dwVisParams).divide(255);
  var probabilityHillshade = rgbImage.multiply(hillshade);

  return probabilityHillshade;
};


// Calculate burned area within fire perimeter, in acres
var getFireArea = function(fireDict,affected_areas) {
  var burn_lc = landcover.expression('(lc >= 5 & lc <= 7)', {lc: landcover}).selfMask();

  return burn_lc.rename('area').reduceRegion({
      reducer: ee.Reducer.count().unweighted(),
      geometry: affected_areas.geometry(),
      crs: nlcdProj,
      scale: nlcdProj.nominalScale(),
      tileScale: fireDict.tileScale,
      bestEffort: true
     }).getNumber('area')
      .multiply(30*30*0.000247105).round();
};

var indexJoin = function(collectionA, collectionB, propertyName) {
  var joined = ee.ImageCollection(ee.Join.saveFirst(propertyName).apply({
    primary: collectionA,
    secondary: collectionB,
    condition: ee.Filter.equals({
      leftField: 'system:index',
      rightField: 'system:index'})
  }));

  return joined.map(function(image) {
    return image.addBands(ee.Image(image.get(propertyName)));
  });
};

var NBRwithCloudsMasked = function(image) {
  var cloudProb = image.select('probability');
  var isCloud = cloudProb.gt(65);
  return image.normalizedDifference(['B8','B12']).rename('NBR')
    .updateMask(isCloud.not())
    .copyProperties(image,['system:time_start']);
};

var getBurnSeverity = function(fireDict, area_of_interest, fireConf) {

  var dnbr_dates = {
    'preS': ee.Date(fireDict.start).advance(-16,'day'),
    'preE': ee.Date(fireDict.start).advance(-2,'day'),
    'postS': ee.Date(fireDict.end),
    'postE': ee.Date(fireDict.end).advance(15,'day')
  };

  var s2SR_aoi = s2SR.select(['B8','B12'])
    .filterBounds(area_of_interest)
    .filterDate(dnbr_dates.preS,dnbr_dates.postE);

  var s2c_aoi = s2c.filterBounds(area_of_interest)
    .filterDate(dnbr_dates.preS,dnbr_dates.postE);

  var s2NBRcol = indexJoin(s2SR_aoi,s2c_aoi,'probability')
    .map(NBRwithCloudsMasked);

  var preFire = s2NBRcol.filterDate(dnbr_dates.preS,dnbr_dates.preE).max();
  var postFire = s2NBRcol.filterDate(dnbr_dates.postS,dnbr_dates.postE).min();

  var dNBR = preFire.subtract(postFire).rename('dNBR');

  dNBR = dNBR.expression('(dNBR > 0.15 & dNBR <= 0.27) + (dNBR > 0.27 & dNBR <= 0.44)*2' +
    '+ (dNBR > 0.44 & dNBR <= 0.66)*3 + (dNBR > 0.66 & dNBR <= 1.3)*4',
    {dNBR: dNBR});

  var dNBRmask;
  if (fireDict.tileScale < 8) {
    dNBRmask = dNBR.selfMask().reduceNeighborhood({
      reducer: ee.Reducer.count(),
      kernel: ee.Kernel.square(1),
      optimization:'boxcar'
    }).gt(4);
  } else {
     dNBRmask = dNBR.selfMask();
  }

  return dNBR.updateMask(dNBRmask)
    .updateMask(landcover.expression('(lc >= 3 & lc <= 7)', {lc: landcover}))
    .updateMask(fireConf.gt(0.25));
};

var getFireAreaSeverity = function(fireDict,burnSeverity,area_of_interest) {

  var fireAreaDict;
  if (fireDict.severityArea === undefined) {
    var fireAreaHist = burnSeverity
      .reduceRegion({
        reducer: ee.Reducer.fixedHistogram(1,5,4).unweighted(),
        geometry: area_of_interest,
        crs: nlcdProj,
        scale: nlcdProj.nominalScale(),
        tileScale: fireDict.tileScale,
        bestEffort: true
       }).toArray().slice(1,1,2).project([0]);

    fireAreaDict = ee.Dictionary({
      'All': fireAreaHist.accum(0).get([3]),
      'Low': fireAreaHist.get([0]),
      'Moderate-Low': fireAreaHist.get([1]),
      'Moderate-High': fireAreaHist.get([2]),
      'High': fireAreaHist.get([3])
    });

    fireAreaDict = fireAreaDict.map(function(x,y) {
      return ee.Number(y).multiply(30*30*0.000247105).round()});

    return ee.FeatureCollection([
      ee.Feature(null, {
        'ID': '',
        'Low': fireAreaDict.get('Low'),
        'Moderate-Low': fireAreaDict.get('Moderate-Low'),
        'Moderate-High': fireAreaDict.get('Moderate-High'),
        'High': fireAreaDict.get('High')
      }),
      ee.Feature(null, {
        'ID': '',
        'Total': fireAreaDict.get('All')
      }),
    ]);
  } else {
    fireAreaDict = fireDict.severityArea;
    return ee.FeatureCollection([
      ee.Feature(null, {
        'ID': '',
        'Low': fireAreaDict[0],
        'Moderate-Low': fireAreaDict[1],
        'Moderate-High': fireAreaDict[2],
        'High': fireAreaDict[3]
      }),
      ee.Feature(null, {
        'ID': '',
        'Total': fireAreaDict[4]
      }),
    ]);
  }
};

var getFireProgression = function(fireDict,area_of_interest) {
  var nHour = ee.Date(fireDict.end).difference(fireDict.start,'hour').floor();

  var fireAreaByHour = ee.List.sequence(1,nHour,fireDict.timeInterval)
    .map(function(iStep) {
      var stTime = fireDict.start;
      var endTime = stTime.advance(iStep,'hour');

      var smoothed_confidence = getFireConf(area_of_interest,fireDict,endTime);
      var affected_areas99 = getFirePerim(fireDict,area_of_interest,smoothed_confidence,0.99);
      var affected_areas90 = getFirePerim(fireDict,area_of_interest,smoothed_confidence,0.9);
      var affected_areas75 = getFirePerim(fireDict,area_of_interest,smoothed_confidence,0.75);

      return ee.Feature(affected_areas90.geometry(), {
          timeStep: iStep,
          P75: getFireArea(fireDict,affected_areas75),
          P90: getFireArea(fireDict,affected_areas90),
          P99: getFireArea(fireDict,affected_areas99),
        });
    });

  return ee.FeatureCollection(fireAreaByHour);
};

var chartActiveFires = function(fireDict,area_of_interest,localStartTime) {

  // Satellite data
  var goes16_data = goes16fdcf.select('Mask');
  var goes17_data = goes17fdcf.select('Mask');

  var goes16proj = goes16_data.first().projection();
  var goes17proj = goes17_data.first().projection();

  var getPixCount = function(goes16,goes17) {
    var goes16_pixels = ee.Number(ee.Algorithms.If(
      goes16.size().gt(0),
      goes16.min().lte(35).rename('pixels')
      .reduceRegion({
        geometry: area_of_interest,
        reducer: ee.Reducer.sum().unweighted(),
        crs: goes16proj,
        scale: goes16proj.nominalScale(),
      }).getNumber('pixels'),0));

    var goes17_pixels = ee.Number(ee.Algorithms.If(
      goes17.size().gt(0),
      goes17.min().lte(35).rename('pixels')
      .reduceRegion({
        geometry: area_of_interest,
        reducer: ee.Reducer.sum().unweighted(),
        crs: goes17proj,
        scale: goes17proj.nominalScale(),
      }).getNumber('pixels'),0));

    return ee.Feature(null,{'GOES-16':goes16_pixels, 'GOES-17':goes17_pixels,
      'GOES': ee.List([goes16_pixels,goes17_pixels]).reduce(ee.Reducer.max())
    });
  };

  var nHour = ee.Date(fireDict.end).difference(fireDict.start,'hour').floor();

  var activeFiresByHour = ee.List.sequence(1,nHour,1).map(function(iStep) {
    var stTime = ee.Date(fireDict.start)
      .advance(ee.Number(iStep).subtract(1),'hour');
    var endTime = stTime.advance(1,'hour');

    var goes16hr = goes16_data.filterDate(stTime,endTime);
    var goes17hr = goes17_data.filterDate(stTime,endTime);

    return getPixCount(goes16hr,goes17hr).set('timeStep',iStep);
  });

  return ui.Chart.feature.byFeature({
    features: ee.FeatureCollection(activeFiresByHour),
    xProperty: 'timeStep',
    yProperties: ['GOES']
  }).setOptions({
      title: 'Active Fire Pixels',
      titleTextStyle: {fontSize: '15'},
    vAxis: {
      title: 'Maximum Count',
      titleTextStyle: {fontSize: '12'},
    },
    hAxis: {
      title: 'Hours Since ' + localStartTime + 'h (LT)',
      titleTextStyle: {fontSize: '12'},
    },
    legend: {position: 'none'},
    series: {
      0: {color: '#B03A2E', lineWidth: 1.5}
    }
  });
};

var getMTBStable = function(area_of_interest) {
  var maxYear = ee.Number.parse(fireYrSelect.getValue());

  var MTBSregion = MTBS.filterBounds(area_of_interest)
    .filter(ee.Filter.stringStartsWith('Fire_Type','Wildfire'))
    .filter(ee.Filter.lt('Year',maxYear))
    .sort('Year',false);


  var capitalize = function(string) {
    var splitString = ee.String(string).split(' ');
    var capString = splitString.map(function(x) {
      return ee.String(x).slice(0,1)
        .cat(ee.String(x).slice(1,100).toLowerCase());
    });
    return capString.iterate(function(x,first) {
      return ee.String(first).cat(' ').cat(ee.String(x))},ee.String(''));
  };

  var fireTable = ee.Algorithms.If(MTBSregion.size().gt(0),
    MTBSregion.select(['Fire_Name','Year','Acres'],['Fire Name','Year','Acres'])
      .map(function(feature) {
        return feature.set('Year',feature.getNumber('Year').toInt().format())
          .set('Fire Name',capitalize(feature.getString('Fire Name')));
      }),
    ee.FeatureCollection([ee.Feature(null, {'Fire Name':'','Year':'','Acres':''})]));

  return ui.Chart.feature.byFeature(ee.FeatureCollection(fireTable),
    'Fire Name',['Year','Acres']).setChartType('Table');
};

var displayFire = function(fireName) {

  var year = fireYrSelect.getValue();
  var fireParamsYrList = fireParamsList[year];

  var Map1 = ui.Map().setOptions('TERRAIN'); Map1.style().set({cursor:'crosshair'});
  var Map2 = ui.Map().setOptions('TERRAIN'); Map2.style().set({cursor:'crosshair'});

  Map1.unlisten();

  var controlPanel = ui.Panel({
    layout: ui.Panel.Layout.flow('horizontal'),
  });

  var infoWrapper = ui.Panel(
    {style: {width: '205px', maxHeight: '80%', position: 'bottom-right'}
  });

  var mapPanel = ui.Panel({
    layout: ui.Panel.Layout.flow('horizontal'),
  });

  var map_panels = ui.SplitPanel({
    firstPanel: ui.Panel({widgets: [Map2], style: {width: '0%'}}),
    secondPanel: ui.Panel({widgets: [Map1], style: {Width: '100%'}}),
  });

  var mapGrid = ui.Panel([map_panels],
    ui.Panel.Layout.Flow('horizontal'), {stretch: 'both'}
  );

  // Fire Parameters
  var fireDict = fireParamsYrList[fireName];
  var area_of_interest = fireDict.AOI;

  var localStartTime = UTCtoLT(fireDict.start,timeZoneList[fireDict.state]).getInfo();

  var smoothed_confidence = getFireConf(area_of_interest,fireDict,fireDict.end);
  var affected_areas = getFirePerim(fireDict,area_of_interest,smoothed_confidence,0.9);
  var affected_areas_smoothed_outline = getFirePerimImg(affected_areas,3);

  var affected_areas75 = getFirePerim(fireDict,area_of_interest,smoothed_confidence,0.75);
  var affected_areas99 = getFirePerim(fireDict,area_of_interest,smoothed_confidence,0.99);
  var affected_areas_smoothed_outlineBounds =
    getFirePerimImg(affected_areas99.merge(affected_areas75),1.5);

  var fireAreaByHour = getFireProgression(fireDict,area_of_interest);

  var title = ui.Label(fireName + ' Fire | ' + fireDict.stDate + ' | ' + fireDict.status,
    {stretch: 'horizontal', textAlign: 'left', fontWeight: 'bold',
      fontSize: '20px', color: 'red', margin: '0 8px 8px 15px'});

  // Map layers
  ui.root.add(title); ui.root.add(mapGrid);
  ui.root.setLayout(ui.Panel.Layout.Flow('vertical'));

  Map1.setControlVisibility({mapTypeControl: false, layerList: false});
  Map1.addLayer(smoothed_confidence.selfMask(),
    {opacity: 0.4, min: 0, max: 1, palette: fire_confidence_palette},
      'Smoothed confidence');

  Map1.addLayer(affected_areas_smoothed_outline,
    {palette: 'black'},'Smoothed affected areas, 90%', true, 0.75);
  Map1.addLayer(affected_areas_smoothed_outlineBounds,
    {palette: '#D3D3D3'},'Smoothed affected areas, 75%-99%', true, 0.75);

  Map2.setControlVisibility({layerList: false, mapTypeControl: false});

  Map2.addLayer(landcover.selfMask(), {min: 1, max: 9, palette: lc_palette},
    'NLCD Land Cover', true, 0.5);


  var fireAreaSeverity = null;
  if (fireDict.status == 'Final') {
    var dynamicWorld = getDynamic(fireDict,area_of_interest);

    Map2.addLayer(dynamicWorld, {min: 0, max: 0.8},
      'Dynamic World Land Cover', false, 0.75);

    var burnSeverity = getBurnSeverity(fireDict,area_of_interest,smoothed_confidence);
    fireAreaSeverity = getFireAreaSeverity(fireDict,burnSeverity,area_of_interest);

    Map2.addLayer(burnSeverity, {min: 1, max: 4, palette: burn_severity_palette},
      'Burn severity', false, 0.75);
  }

  Map2.addLayer(affected_areas_smoothed_outline,
    {palette: 'black'},'Smoothed affected areas', true, 0.75);

  var legendDict = {
    'NLCD Land Cover': getLegendDiscrete('Land Cover',lc_labels,lc_palette,
      'Data: USGS National Land Cover Database (30-m)'),
    'Dynamic World Land Cover': getLegendDiscrete('Land Cover',dynamic_labels,dynamic_palette,
      'Data: Google/WRI Dynamic World v1 (10-m)'),
    'Burn Severity': getLegendDiscrete('Burn Severity',burn_severity_labels,burn_severity_palette,
      'Data: Sentinel-2A (10-m) surface reflectance'),
  };

  var layerNames = ['NLCD Land Cover','Dynamic World Land Cover', 'Burn Severity'];
  if (fireDict.status == 'Active') {
    layerNames = ['Land Cover'];
  }

  var layerDict = {
    'NLCD Land Cover': [true,false,false],
    'Dynamic World Land Cover': [false,true,false],
    'Burn Severity': [false,false,true],
  };

  var legendParent = ui.Panel({style: {margin:'0', padding: '0'}});
  var layerSelect = ui.Select({
    items: layerNames,
    value: 'NLCD Land Cover',
    style: {stretch:'horizontal'}
  });

  var layerPanel = ui.Panel({
    widgets: [layerSelect,legendParent],
    style: {width: '150px',margin:'0',padding:'0 0 8px 0', position:'bottom-left'}
  });

  legendParent.add(legendDict['NLCD Land Cover']);

  layerSelect.onChange(function(selected) {
    legendParent.clear();
    legendParent.add(legendDict[selected]);

    for (var layer = 0; layer <= 2; layer++) {
      var mapLayer = Map2.layers().get(layer);
      mapLayer.setShown(layerDict[selected][layer]);
    }
  });

  Map2.add(layerPanel);

  var fireArea = [
    getFireArea(fireDict,affected_areas),
    getFireArea(fireDict,affected_areas99),
    getFireArea(fireDict,affected_areas75)
  ];

  var infoPanel = getInfoPanel(Map1,fireDict,fireName,fireArea,fireAreaSeverity);

  var hideMode = true;
  var hideShowButton = ui.Button({
    label: 'Hide Info Panel',
    onClick: function() {
      hideMode = !hideMode;
      hideShowButton.setLabel(hideMode ? 'Hide Info Panel': 'Show Info Panel');
      if (!hideMode) {
        infoWrapper.style().set({width: '115px', height: '45px'});
      } else {
        infoWrapper.style().set({width: '205px', height: 'auto'});
      }
    },
      style: {padding: '0', margin: '0'}
  });

  var MTBSwrapper = ui.Panel({style: {margin: '0 0 0 -8px', padding: '0'}});

  var hideModeMTBS = true;
  var hideShowButtonMTBS = ui.Button({
    label: 'See Historical Fires',
    onClick: function() {
      hideModeMTBS = !hideModeMTBS;
      hideShowButtonMTBS.setLabel(hideModeMTBS ? 'See Historical Fires': 'Hide Historical Fires');
      var MTBSpanel = getMTBStable(area_of_interest);

      if (!hideModeMTBS) {
        MTBSwrapper.add(MTBSpanel);
      } else {
        MTBSwrapper.clear();
      }
    },
      style: {padding: '0', margin: '0'}
  });

  infoPanel.add(hideShowButtonMTBS)
    .add(ui.Label('Data: Monitoring Trends in Burn Severity (MTBS) from 1984-2019',
      {fontSize:'12px', margin:'0', padding:'5px 12px 0 0', color:'#777'}))
    .add(MTBSwrapper);

  infoWrapper.add(hideShowButton).add(infoPanel);
  Map1.add(infoWrapper);

  var chartWrapper = ui.Panel({
    widgets: [],
    style: {position:'bottom-left', width:'400px', maxHeight: '90%'}
  });

  var chartPanel = ui.Panel({
    widgets: [],
    style: {margin: '8px -8px -8px -8px', width:'400px'}
  });

  var hideChartMode = true;
  var hideShowChartButton = ui.Button({
    label: 'Hide Charts',
    onClick: function() {
      hideChartMode = !hideChartMode;
      hideShowChartButton.setLabel(hideChartMode ? 'Hide Charts': 'Show Charts');
      if (!hideChartMode) {
        chartWrapper.style().set({width: '97px', height: '45px'});
      } else {
        chartWrapper.style().set({width: '400px', height: 'auto'});
      }
    },
      style: {padding: '0', margin: '0'}
  });

  var fireProgAnim = getFireProgImgAnim(area_of_interest,fireAreaByHour);
  var fireProgressionChart = chartFireProgression(fireDict,fireAreaByHour,localStartTime);
  var activeFireChart = chartActiveFires(fireDict,area_of_interest,localStartTime);
  chartPanel.add(fireProgressionChart).add(fireProgAnim).add(activeFireChart);
  chartWrapper.add(hideShowChartButton).add(chartPanel);

  Map1.add(chartWrapper);

  ui.root.onResize(function(deviceInfo) {
    if (deviceInfo.is_desktop & deviceInfo.width > 900) {
      chartWrapper.style().set({width: '400px', height: 'auto'});
      hideShowChartButton.setLabel('Hide Charts');
      hideChartMode = true;
    } else {
      chartWrapper.style().set({width: '97px', height: '45px'});
      hideShowChartButton.setLabel('Show Charts');
      hideChartMode = false;
    }
  });

  ui.Map.Linker([Map1,Map2]);

  Map1.centerObject(area_of_interest,fireDict.zoom);

  ui.url.set('year',year);
  ui.url.set('fire_name',fireName);
};
