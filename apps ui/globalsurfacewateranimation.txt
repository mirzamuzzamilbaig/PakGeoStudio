function forwardBack(col) {
  return col.merge(col.sort('year', false));
}

// ### Ghost of stream channel path
function fadeAnimation() {
  var fadeFilter = ee.Image(1).visualize({palette: bgColor,
    opacity: ((fadeSlider.getValue() > 0.98) ? 0.98 : fadeSlider.getValue())});

  var fadeFilterCol = col.map(function(img) {
    var imgVis = img.selfMask().visualize({palette: riverColor});
    return imgVis.blend(fadeFilter).set('year', img.get('year'));
  });

  var fadeCol = ee.ImageCollection.fromImages(yearSeq.map(function(year) {
    var fadeComp =
        fadeFilterCol.filter(ee.Filter.lte('year', year)).sort('year').mosaic();
    var thisYearImg = col.filter(ee.Filter.eq('year', year)).first().selfMask().visualize({
      palette: riverColor
    });
    return bgImg.blend(fadeComp).blend(thisYearImg).set('year', year);
  }));
  return fadeCol;
}

function cumulativeAnimation() {
  var cumulativeFilterCol = col.map(function(img) {
    return img.selfMask().visualize(visLayer).set('year', img.get('year'));
  });

  var cumulativeCol = ee.ImageCollection.fromImages(yearSeq.map(function(year) {
    var cumulativeComp = cumulativeFilterCol.filter(ee.Filter.lte('year', year))
                             .sort('year')
                             .mosaic();
    return bgImg.blend(cumulativeComp).set('year', year);
  }));
  return cumulativeCol;
}

// ### Control panel call back functions
function clearMap() {
  if(dirty) {
    Map.remove(gifPanel);
    clearGeometry();
    dirty = false;
  }
}
clearButton.onClick(clearMap);

function clearGeometry(){
  var layers = drawingTools.layers();
  layers.get(0).geometries().remove(layers.get(0).geometries().get(0));
}

function drawRectangle(){
  if(dirty) Map.remove(gifPanel);
  dirty = false;
  clearGeometry();
  drawingTools.setShape('rectangle');
  drawingTools.draw();
}
drawButton.onClick(drawRectangle);

function switchAni() {
  render = selectButton.getValue();
  ui.url.set('render', render);
  if(render == 'Fading history') {
    fadeSlider.setDisabled(false);
  } else {
    fadeSlider.setDisabled(true);
  }
  if(dirty) renderAnimations();
}
selectButton.onChange(switchAni);

// Function to plot chart on drawing events.
var animation;
var gifPanel;
var dirty = false;
function renderAnimations() {
  if(dirty) Map.remove(gifPanel);

  aoi = drawingTools.layers().get(0).getEeObject();

  ui.url.set('aoi', aoi.toGeoJSONString());
  ui.url.set('fps', fpsSlider.getValue());
  ui.url.set('fade', fadeSlider.getValue());

  drawingTools.setShape(null);

  var videoArgs = {
    dimensions: height,
    region: aoi,
    framesPerSecond: fpsSlider.getValue(),
    crs: 'EPSG:3857'
  };

  if(render == 'Fading history') {
    animation = ui.Thumbnail({image: forwardBack(fadeAnimation()), params: videoArgs,
      style: {margin: '3px'}});
  } else {
    animation = ui.Thumbnail({image: forwardBack(cumulativeAnimation()), params: videoArgs,
      style: {margin: '3px'}});
  }

  gifPanel = ui.Panel({style: {position: 'bottom-right', padding: '1px',
    backgroundColor: 'rgba(0, 0, 0, 0.5)'}});
  gifPanel.add(animation);
  Map.add(gifPanel);
  dirty = true;
}
drawingTools.onEdit(ui.util.debounce(renderAnimations, 500));
drawingTools.onDraw(ui.util.debounce(renderAnimations, 500));

// Initialize animation.
renderAnimations();
